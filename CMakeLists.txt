cmake_minimum_required(VERSION 3.21)
project(KenvixBluetoothUnlock LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kenvix_protocol STATIC core/protocol.cpp core/remote_protocol.cpp)
target_include_directories(kenvix_protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core)
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE kenvix_protocol)
  add_test(NAME protocol_tests COMMAND protocol_tests)
  add_executable(remote_protocol_tests tests/remote_protocol_tests.cpp)
  target_link_libraries(remote_protocol_tests PRIVATE kenvix_protocol)
  add_test(NAME remote_protocol_tests COMMAND remote_protocol_tests)
endif()
if(WIN32)
  add_subdirectory(windows)
endif()
