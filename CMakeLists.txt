cmake_minimum_required(VERSION 3.21)
project(BluetoothUnlock LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bluetooth_unlock_protocol STATIC core/protocol.cpp core/remote_protocol.cpp core/ble_presence.cpp)
target_include_directories(bluetooth_unlock_protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core)
include(CTest)
if(BUILD_TESTING)
  add_executable(ble_presence_tests tests/ble_presence_tests.cpp)
  target_link_libraries(ble_presence_tests PRIVATE bluetooth_unlock_protocol)
  target_compile_options(ble_presence_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>)
  add_test(NAME ble_presence_tests COMMAND ble_presence_tests)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE bluetooth_unlock_protocol)
  add_test(NAME protocol_tests COMMAND protocol_tests)
  add_executable(remote_protocol_tests tests/remote_protocol_tests.cpp)
  target_link_libraries(remote_protocol_tests PRIVATE bluetooth_unlock_protocol)
  add_test(NAME remote_protocol_tests COMMAND remote_protocol_tests)
  if(NOT WIN32)
    find_package(OpenSSL QUIET)
    if(OpenSSL_FOUND)
      add_executable(interop_openssl_tests tests/interop_openssl_tests.cpp)
      target_link_libraries(interop_openssl_tests PRIVATE bluetooth_unlock_protocol OpenSSL::Crypto)
      add_test(NAME interop_openssl_tests COMMAND interop_openssl_tests)
      set_tests_properties(interop_openssl_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
    endif()
  endif()
endif()
if(WIN32)
  add_subdirectory(windows)
endif()
